# MIPS program to compute (a + b) - c

.text main: # Read input a li $v0, 5 # System call: read integer syscall move $t0, $v0 # Store a in $t0 # Read input b li $v0, 5 syscall move $t1, $v0 # Store b in $t1 # Read input c li $v0, 5 syscall move $t2, $v0 # Store c in $t2 # Compute (a + b) - c add $t3, $t0, $t1 # t3 = a + b sub $t3, $t3, $t2 # t3 = (a + b) - c # Print result move $a0, $t3 li $v0, 1 # System call: print integer syscall # Exit li $v0, 10 syscall
```

# MIPS program to compute sum of array [1, 2, 3, 4, 5]

.data arr: .word 1, 2, 3, 4, 5 .text main: la $t0, arr # Load array address li $t1, 5 # Array length li $t2, 0 # Sum = 0 li $t3, 0 # Index = 0 loop: bge $t3, $t1, end # If index >= length, exit lw $t4, 0($t0) # Load arr[index] add $t2, $t2, $t4 # Sum += arr[index] addi $t0, $t0, 4 # Move to next element addi $t3, $t3, 1 # Index++ j loop end: move $a0, $t2 li $v0, 1 # System call: print integer syscall li $v0, 10 # System call: exit syscall
```
